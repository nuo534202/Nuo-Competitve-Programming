# LeetCode 11 - 20

## 11. 盛水最多的容器

使用双指针法，分别从数组的两端开始向中间移动。每次计算当前指针所指的高度和它们之间的距离，更新最大面积。然后根据较小的高度来决定移动哪个指针。

时间复杂度： $O(n)$。

```cpp
class Solution {
public:
    int maxArea(vector<int>& height) {
        int l = 0, r = height.size() - 1, ans = 0;
        while (l < r)
        {
            ans = max(ans, min(height[l], height[r]) * (r - l));
            if (height[l] > height[r]) r--;
            else l++;
        }
        return ans;
    }
};
```

## 12. 整数转罗马数字

使用预定义的字符串数组来表示不同位数的罗马数字。根据输入整数的位数，直接拼接对应的字符串即可。

时间复杂度： $O(1)$。

```cpp
class Solution {
    const string thousands[4] = {"", "M", "MM", "MMM"};
    const string hundreds[10] = {"", "C", "CC", "CCC", "CD", "D", "DC", "DCC", "DCCC", "CM"};
    const string tens[10] = {"", "X", "XX", "XXX", "XL", "L", "LX", "LXX", "LXXX", "XC"};
    const string ones[10] = {"", "I", "II", "III", "IV", "V", "VI", "VII", "VIII", "IX"};

public:
    string intToRoman(int num) {
        return thousands[num / 1000] + hundreds[num % 1000 / 100] + tens[num % 100 / 10] + ones[num % 10];
    }
};
```

## 13. 罗马数字转整数

使用一个哈希表来存储罗马数字及其对应的整数值。遍历输入字符串，如果当前字符对应的值小于下一个字符对应的值，则减去当前字符的值，否则加上当前字符的值。

时间复杂度： $O(n)$。

```cpp
class Solution {
private:
    unordered_map<char, int> symbolValues = {
        {'I', 1},
        {'V', 5},
        {'X', 10},
        {'L', 50},
        {'C', 100},
        {'D', 500},
        {'M', 1000},
    };

public:
    int romanToInt(string s) {
        int ans = 0;
        int n = s.length();
        for (int i = 0; i < n; ++i) {
            int value = symbolValues[s[i]];
            if (i < n - 1 && value < symbolValues[s[i + 1]]) {
                ans -= value;
            } else {
                ans += value;
            }
        }
        return ans;
    }
};
```

## 14. 最长公共前缀

扫描每个字符串的每一位，直到遇到不相同的字符为止。使用一个循环遍历每个字符串的字符，如果所有字符串在该位置的字符都相同，则将该字符加入结果中，否则返回当前结果。

时间复杂度： $O(mn)$。

```cpp
class Solution {
public:
    string longestCommonPrefix(vector<string>& strs) {
        string longest;
        for (int j = 0; j < (int)strs[0].size(); j++)
        {
            int i;
            for (i = 0; i < (int)strs.size(); i++)
            {
                if (j > (int)strs[i].size() || strs[0][j] != strs[i][j])
                    return longest;
            }
            if (i == (int)strs.size())
                longest.push_back(strs[0][j]);
        }
        return longest;
    }
};
```

## 15. 三数之和

**双指针**：枚举第一个数 $i$，然后使用双指针在剩下的数组中寻找两个数，使得它们的和为 $-nums[i]$。为了避免重复解，需要在枚举第一个数时跳过相同的数，并在找到一个解后跳过相同的第二个和第三个数。

时间复杂度： $O(n^2)$。

```cpp
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        sort(nums.begin(), nums.end());
        int n = nums.size();
        vector<vector<int>> ans;
        for (int i = 0; i + 2 < n; i++)
        {
            if (i && nums[i] == nums[i - 1])
                continue;
            if (nums[i] > 0 || nums[i] + nums[i + 1] + nums[i + 2] > 0)
                break;
            if (nums[i] + nums[n - 2] + nums[n - 1] < 0)
                continue;
            int l = i + 1, r = n - 1;
            while (l < r)
            {
                int sum = nums[l] + nums[r];
                if (sum == -nums[i])
                {
                    ans.push_back({nums[i], nums[l], nums[r]});
                    l++, r--;
                    while (l < r && nums[l] == nums[l - 1])
                        l++;
                    while (l < r && nums[r] == nums[r + 1])
                        r--;
                }
                else if (sum < -nums[i])
                    l++;
                else
                    r--;
            }
        }
        return ans;
    }
};
```

## 16. 最接近的三数之和

**双指针**：枚举第一个数 $i$，然后使用双指针在剩下的数组中寻找两个数，使得它们的和与目标值 $target$ 的差最小。

时间复杂度： $O(n^2)$。

```cpp
class Solution {
public:
    int threeSumClosest(vector<int>& nums, int target) {
        sort(nums.begin(), nums.end());
        int n = nums.size();
        int ans = -1e6;
        for (int i = 0; i + 2 < n; i++)
        {
            if (i && nums[i] == nums[i - 1])
                continue;
            int l = i + 1, r = n - 1;
            while (l < r)
            {
                int sum = nums[l] + nums[r];
                if (sum + nums[i] == target)
                    return target;
                else if (nums[i] + sum < target)
                {
                    if (target - nums[i] - sum < abs(target - ans))
                        ans = nums[i] + sum;
                    l++;
                }
                else
                {
                    if (nums[i] + sum - target < abs(target - ans))
                        ans = nums[i] + sum;
                    r--;
                }
            }
        }
        return ans;
    }
};
```

## 17. 电话号码的字母组合

先定义一个从数字到字母的映射表，然后遍历给出的数字字符串，将当前数字字符所有可能的字母跟之前已经组合好的字符串进行拼接，形成当前位置下所有可能的答案组合，最后返回所有组合。

时间复杂度： $O(n \times 4^n)$

```cpp
class Solution {
public:
    vector<vector<char>> digit_char_map = 
    {
        { 'a', 'b', 'c' }, { 'd', 'e', 'f' },
        { 'g', 'h', 'i' }, { 'j', 'k', 'l' }, { 'm', 'n', 'o' },
        { 'p', 'q', 'r', 's' }, { 't', 'u', 'v' }, { 'w', 'x', 'y', 'z' }
    };
    vector<string> letterCombinations(string digits) {
        vector<string> ans = { "" };
        for (char c : digits)
        {
            vector<string> tmp = ans;
            ans.clear();
            for (char ch : digit_char_map[c - '2'])
            {
                for (const string& str : tmp)
                {
                    ans.push_back(str + ch);
                }
            }
        }
        return ans;
    }
};
```

## 18. 四数之和

排序 + 双指针，做法跟前面几题类似。

时间复杂度： $O(n^3)$。

```cpp
class Solution {
public:
    vector<vector<int>> fourSum(vector<int>& nums, int target) {
        vector<vector<int>> ans;
        sort(nums.begin(), nums.end());
        int n = nums.size();
        for (int i = 0; i + 3 < n; i++)
        {
            if (i && nums[i] == nums[i - 1])
                continue;
            for (int j = i + 1; j + 2 < n; j++)
            {
                if (j > i + 1 && nums[j] == nums[j - 1])
                    continue;
                int l = j + 1, r = n - 1;
                while (l < r)
                {
                    long long sum = (long long)nums[i] + nums[j] + nums[l] + nums[r];
                    if (sum == target)
                    {
                        ans.push_back({ nums[i], nums[j], nums[l], nums[r] });
                        l++, r--;

                        while (l < r && nums[l] == nums[l - 1])
                            l++;
                        while (l < r && nums[r] == nums[r + 1])
                            r--;
                    }
                    else if (sum < target)
                        l++;
                    else
                        r--;
                }
            }
        }
        return ans;
    }
};
```

## 19. 删除链表的倒数第 N 个结点

使用递归的方式，从链表的末尾开始计算节点的个数，当到达倒数第 N 个节点时，将其删除。

时间复杂度： $O(n)$。

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    void removeNode(ListNode *head, int n, int &cnt)
    {
        if (head->next == nullptr)
        {
            cnt = 1;
            return;
        }
        removeNode(head->next, n, cnt);
        cnt++;
        if (n + 1 == cnt)
        {
            ListNode *remove_node = head->next;
            head->next = head->next->next;
            delete(remove_node);
        }
    }

    ListNode* removeNthFromEnd(ListNode* head, int n) {
        int cnt = 0;
        ListNode *tmp = new ListNode();
        tmp->next = head;
        removeNode(tmp, n, cnt);
        return tmp->next;
    }
};
```

## 20. 有效的括号

用一个栈来判断括号序列是否合法，遇到左半边的括号时就直接将括号放入栈，遇到又半边的括号时判断括号是否匹配，如果匹配就将栈顶元素弹出，否则返回 false。最后判断栈是否为空，如果为空则说明括号序列合法。

时间复杂度： $O(n)$。

```cpp
class Solution {
public:
    bool isValid(string s) {
        stack<char> st;
        for (char c : s)
        {
            if (c == '(' || c == '[' || c == '{')
                st.push(c);
            else
            {
                if (st.empty())
                    return false;
                if ((c == ')' && st.top() == '(') ||
                    (c == ']' && st.top() == '[') || 
                    (c == '}' && st.top() == '{'))
                    st.pop();
                else
                    return false;
            }
        }
        if (st.empty())
            return true;
        else
            return false;
    }
};
```
